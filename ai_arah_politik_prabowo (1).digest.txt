import numpy as np
import pandas as pd

# 1. Set up daily date range
date_range = pd.date_range(start='2020-01-01', end='2026-12-31', freq='D')
n_days = len(date_range)

# Set seed for local generation consistency
np.random.seed(SEED)

# Helper function to generate an AR(1) process
def generate_ar1(init_val, phi, sigma, n):
    vals = np.zeros(n)
    vals[0] = init_val
    for t in range(1, n):
      vals[t] = phi * vals[t-1] + np.random.normal(0, sigma)
    return vals

# 2. Continuous time-series features
gdp_growth = 2.5 + generate_ar1(0.0, 0.98, 0.1, n_days) # oscillating around 2.5%
inflation = 3.0 + generate_ar1(0.0, 0.97, 0.15, n_days) # oscillating around 3%
central_bank = 2.0 + 0.5 * inflation + np.random.normal(0, 0.2, n_days) # reacts to inflation

# Political Stability Index [0 to 100]
pol_stability = 70.0 + generate_ar1(0.0, 0.99, 0.8, n_days)
pol_stability = np.clip(pol_stability, 0, 100)

# Public Sentiment Score [-1.0 to 1.0]
sentiment = generate_ar1(0.1, 0.95, 0.05, n_days)
sentiment = np.clip(sentiment, -1.0, 1.0)

# 3. Discrete/Categorical event signals
election_years = date_range.year.isin([2020, 2024, 2026]).astype(int)

# Random policy changes
policy_categories = ['None', 'Tax Cut', 'Tariff Increase', 'Subsidy Bill']
policy_events = np.random.choice(policy_categories, size=n_days, p=[0.95, 0.02, 0.015, 0.015])

# 4. Target trend construction
macro_political_trend = (pol_stability / 100.0) * 0.4 + (sentiment) * 0.3 - (inflation / 10.0) * 0.3

# Compile into DataFrame
df_raw = pd.DataFrame({
    'date': date_range,
    'gdp_growth_annualized': gdp_growth,
    'inflation_rate': inflation,

    'public_sentiment_score': sentiment,
    'election_year': election_years,
    'policy_change_event': policy_events,

def validate_dataset(df):
    print("=== Starting Strict Data Validation ===")
    errors = 0
    warnings = 0

    # 1. Chronological order & index validation
    if not df['date'].is_monotonic_increasing:
        print("[ERROR] The dataset is not sorted chronologically.")
        errors += 1
    else:
        print("[OK] Chronological sorting check passed.")

    # 2. Duplicate dates validation
    duplicate_dates = df['date'].duplicated().sum()
    if duplicate_dates > 0:
        print(f"[ERROR] Found {duplicate_dates} duplicate date entries.")
        errors += 1
    else:
        print("[OK] Duplicate dates check passed.")

    # 3. Missing values check
    missing_counts = df.isnull().sum()
    total_missing = missing_counts.sum()
    if total_missing > 0:
        print(f"[WARNING] Found {total_missing} missing values:")
        print(missing_counts[missing_counts > 0])
        warnings += 1
    else:
        print("[OK] Missing values check passed. No NaNs detected.")

    # 4. Range and constraint validations
    # Political Stability Index [0 to 100]
    if (df['political_stability_index'] < 0).any() or (df['political_stability_index'] > 100).any():
        print("[WARNING] Political Stability Index out of expected [0, 100] bounds.")
        warnings += 1
    else:
        print("[OK] Political Stability Index range check passed.")

    # Public Sentiment Score [-1.0 to 1.0]
    if (df['public_sentiment_score'] < -1.0).any() or (df['public_sentiment_score'] > 1.0).any():
        print("[WARNING] Public Sentiment Score out of expected [-1.0, 1.0] bounds.")
        warnings += 1
    else:
        print("[OK] Public Sentiment Score range check passed.")

    print(f"=== Validation Completed with {errors} Errors and {warnings} Warnings ===\n")
    return errors == 0

import pandas as pd
import numpy as np

# Create a copy to prevent SettingWithCopy warnings
df_clean = df_raw.copy()

# Define horizons
horizons = [7, 30]

for h in horizons:
    # 1. Continuous targets (future value at t + h minus current value at t)
    # To predict t + h using info up to t, the target is future_val - current_val
    df_clean[f'target_reg_{h}d'] = df_clean['macro_political_trend'].shift(-h) - df_clean['macro_political_trend']

    # 2. Categorical targets based on the change
    # We will use thresholds based on quantiles or standard deviation of the changes to label:
    # -1 (Negative Change), 0 (Neutral / Minor Change), 1 (Positive Change)
    change_vals = df_clean[f'target_reg_{h}d'].dropna()
    threshold_up = change_vals.quantile(0.67)
    threshold_down = change_vals.quantile(0.33)

    def label_target(val):
        if pd.isna(val):
            return np.nan
        elif val >= threshold_up:
            return 1
        elif val <= threshold_down:
            return -1
        else:
            return 0

    df_clean[f'target_cls_{h}d'] = df_clean[f'target_reg_{h}d'].apply(label_target)

# Drop the last max(horizons) rows where the targets are NaN due to shifting (no future look-ahead)
max_h = max(horizons)
df_clean = df_clean.iloc[:-max_h].reset_index(drop=True)

print(f"DataFrame shape after removing look-ahead rows: {df_clean.shape}")
for h in horizons:
    print(f"\nDistribution for {h}-day Classification Target:")
    print(df_clean[f'target_cls_{h}d'].value_counts(dropna=False))

df_clean.to_csv('artifacts/cleaned_targeted_data.csv', index=False)

import pandas as pd
import numpy as np

# Load cleaned dataset
df_features = pd.read_csv('artifacts/cleaned_targeted_data.csv')
df_features['date'] = pd.to_datetime(df_features['date'])

# 1. One-hot encode the categorical policy change events
df_features = pd.get_dummies(df_features, columns=['policy_change_event'], prefix='policy', drop_first=True)
# Convert boolean one-hot columns to float
policy_cols = [col for col in df_features.columns if col.startswith('policy_')]
df_features[policy_cols] = df_features[policy_cols].astype(float)

# 2. Define features for lag and rolling calculations
numerical_cols = [
    'gdp_growth_annualized', 'inflation_rate', 'central_bank_rate',
    'political_stability_index', 'public_sentiment_score', 'macro_political_trend'
]

lags = [1, 3, 7]
rolling_windows = [7, 30]

# Calculate lags
for col in numerical_cols:
    for lag in lags:
        df_features[f'{col}_lag_{lag}'] = df_features[col].shift(lag)

# Calculate rolling windows (strictly past-looking with center=False)
for col in numerical_cols:
    for window in rolling_windows:
        # Shift by 1 day first to exclude current day's value if we want to ensure zero look-ahead
        # (though for standard feature engineering, t is known at time t, but shifting ensures t-1 back-window info)
        # We will use the history including today (since targets are t+h, using info up to t is standard)
        df_features[f'{col}_roll_mean_{window}'] = df_features[col].rolling(window=window, center=False).mean()
        df_features[f'{col}_roll_std_{window}'] = df_features[col].rolling(window=window, center=False).std()
        df_features[f'{col}_roll_min_{window}'] = df_features[col].rolling(window=window, center=False).min()
        df_features[f'{col}_roll_max_{window}'] = df_features[col].rolling(window=window, center=False).max()
        # Momentum: difference between current and value 'window' days ago
        df_features[f'{col}_momentum_{window}'] = df_features[col] - df_features[col].shift(window)

# Drop rows with NaNs caused by the rolling/lag calculations (the first max(rolling_windows) rows)
max_window = max(rolling_windows)
df_features = df_features.iloc[max_window:].reset_index(drop=True)

print(f"Feature matrix shape after engineering: {df_features.shape}")
print(f"Number of columns/features: {len(df_features.columns)}")

df_features.to_csv('artifacts/engineered_features.csv', index=False)

import pandas as pd
import numpy as np

def run_leakage_audit(df):
    print("=== Starting Rigorous Data Leakage Audit ===")
    leakage_detected = False

    # 1. Check for target variables in features
    targets = ['target_reg_7d', 'target_cls_7d', 'target_reg_30d', 'target_cls_30d']
    all_columns = list(df.columns)

    # Potential predictors should not contain target names
    potential_predictors = [col for col in all_columns if col not in targets and col != 'date']

    for target in targets:
        if target in potential_predictors:
            print(f"[LEAKAGE ERROR] Target variable '{target}' is present in the predictor list!")
            leakage_detected = True

    # 2. Check for high correlation with future targets in features
    # Any non-target feature with correlation near 1.0 with a future target is suspicious
    for target in targets:
        correlations = df[potential_predictors].corrwith(df[target]).abs()
        high_corr = correlations[correlations > 0.95]
        if len(high_corr) > 0:
            print(f"[LEAKAGE WARNING] Features with suspicious extremely high correlation (>0.95) with '{target}':")
            print(high_corr)
            leakage_detected = True

    # 3. Double-check for look-ahead columns (future metrics, if any, containing raw index or shifting backward)
    # Make sure we don't have negative lag (which indicates looking forward in time)
    negative_lags = [col for col in potential_predictors if 'lag_-' in col or 'lead' in col]
    if len(negative_lags) > 0:
        print(f"[LEAKAGE ERROR] Potential look-ahead features detected: {negative_lags}")
        leakage_detected = True

    # 4. Check for NaNs and Infs
    nan_counts = df[potential_predictors].isnull().sum().sum()
    inf_counts = np.isinf(df[potential_predictors]).sum().sum()

    if nan_counts > 0:
        print(f"[ERROR] Found {nan_counts} NaN values in predictor columns.")
        leakage_detected = True
    if inf_counts > 0:
        print(f"[ERROR] Found {inf_counts} Infinite values in predictor columns.")
        leakage_detected = True

    if not leakage_detected:
        print("[OK] Leakage Audit Passed! No target contamination, look-ahead features, NaNs, or Infs detected.")
    else:
        print("[FAIL] Leakage Audit Failed. Please investigate features.")

    print("=== Leakage Audit Completed ===\n")
    return not leakage_detected

import numpy as np
import pandas as pd
import pickle
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import brier_score_loss, log_loss, accuracy_score, f1_score

# We will build and calibrate a probability ensemble for 7-day and 30-day classification targets.
# Since we are doing multi-class classification (classes 0, 1, 2 representing -1, 0, 1),
# we can train a multiclass Logistic Regression calibrator on the validation set predictions.

calibrated_ensembles = {}

for h, cls_suffix in [(7, '7d'), (30, '30d')]:
    print(f"=== Ensemble & Calibration for {h}-Day Classification Target ===")

    # Retrieve underlying base models
    if h == 7:
        xgb_model = opt_xgb_cls_7d
        lgb_model = lgb_cls_7d # using default lgb for 7d or optimized if kept in memory
    else:
        xgb_model = xgb_cls_30d
        lgb_model = opt_lgb_cls_30d

    y_val = targets[f'target_cls_{cls_suffix}_val'] + 1
    y_test = targets[f'target_cls_{cls_suffix}_test'] + 1

    # Get raw probabilities on validation set
    xgb_val_prob = xgb_model.predict_proba(X_val)
    lgb_val_prob = lgb_model.predict_proba(X_val)
    raw_val_ensemble_prob = (xgb_val_prob + lgb_val_prob) / 2.0

    # Get raw probabilities on test set
    xgb_test_prob = xgb_model.predict_proba(X_test)
    lgb_test_prob = lgb_model.predict_proba(X_test)
    raw_test_ensemble_prob = (xgb_test_prob + lgb_test_prob) / 2.0

    # Define class predictions from raw ensemble
    raw_test_preds = np.argmax(raw_test_ensemble_prob, axis=1)

    # Train Platt Calibrator (multiclass logistic regression) on raw ensemble probabilities
    calibrator = LogisticRegression(C=1.0, solver='lbfgs', multi_class='multinomial', random_state=SEED)
    calibrator.fit(raw_val_ensemble_prob, y_val)

    # Calibrated probabilities on test set
    calibrated_test_prob = calibrator.predict_proba(raw_test_ensemble_prob)
    calibrated_test_preds = np.argmax(calibrated_test_prob, axis=1)

    # Evaluate Log Loss
    raw_log_loss = log_loss(y_test, raw_test_ensemble_prob)
    cal_log_loss = log_loss(y_test, calibrated_test_prob)

    # Evaluate Brier Score (using average of one-vs-rest Brier scores for multiclass)
    def multiclass_brier_score(y_true, prob_matrix):
        # One-hot encode y_true to shape (n_samples, n_classes)
        y_onehot = np.eye(prob_matrix.shape[1])[y_true.astype(int)]
        return np.mean(np.sum((prob_matrix - y_onehot)**2, axis=1))

    raw_brier = multiclass_brier_score(y_test, raw_test_ensemble_prob)
    cal_brier = multiclass_brier_score(y_test, calibrated_test_prob)

    raw_acc = accuracy_score(y_test, raw_test_preds)
    cal_acc = accuracy_score(y_test, calibrated_test_preds)

    print(f"Raw Ensemble:     Log Loss = {raw_log_loss:.5f} | Brier Score = {raw_brier:.5f} | Accuracy = {raw_acc:.2%}")
    print(f"Calibrated Ensem: Log Loss = {cal_log_loss:.5f} | Brier Score = {cal_brier:.5f} | Accuracy = {cal_acc:.2%}\n")

    # Save calibration artifacts
    artifacts = {
        'xgb_model': xgb_model,
        'lgb_model': lgb_model,
        'calibrator': calibrator
    }
    calibrated_ensembles[f'ensemble_{cls_suffix}'] = artifacts

    with open(f'models/ensemble_calibrated_{cls_suffix}.pkl', 'wb') as f:
        pickle.dump(artifacts, f)

    vals = np.zeros(n)
    vals[0] = init_val
    for t in range(1, n):
        vals[t] = phi * vals[t-1] + np.random.normal(0, sigma)
    return vals

import numpy as np
import pandas as pd

def calculate_psi(expected, actual, num_buckets=10):
    """Calculates the Population Stability Index (PSI) between training (expected) and test (actual) distributions."""
    # Determine bin boundaries using quantiles from the expected dataset
    percentiles = np.linspace(0, 100, num_buckets + 1)
    buckets = np.percentile(expected, percentiles)
    buckets[0] = -np.inf
    buckets[-1] = np.inf

    # Calculate counts in each bucket
    expected_counts, _ = np.histogram(expected, bins=buckets)
    actual_counts, _ = np.histogram(actual, bins=buckets)

    # Convert to fractions with small constant to prevent division by zero
    expected_pct = np.where(expected_counts == 0, 0.0001, expected_counts) / len(expected)
    actual_pct = np.where(actual_counts == 0, 0.0001, actual_counts) / len(actual)

    # Compute PSI
    psi_value = np.sum((actual_pct - expected_pct) * np.log(actual_pct / expected_pct))
    return psi_value

# Select main baseline features to audit for drift
key_drift_features = [
    'macro_political_trend',

def predict_political_direction(raw_input_df):
    """
    Production-grade inference pipeline function.
    Takes raw daily macroeconomic & political features, reconstructs engineered features
    with historical window alignment, and predicts 7-day policy shift probabilities (calibrated)
    along with 30-day continuous macro-political trends.
    """
    # 1. Feature Engineering & Generation matching raw structure
    processed_df = raw_input_df.copy()
    processed_df['date'] = pd.to_datetime(processed_df['date'])

    # Map and restore policy dummies
    for category in ['Subsidy Bill', 'Tariff Increase', 'Tax Cut']:
        col_name = f'policy_{category}'
        processed_df[col_name] = (processed_df['policy_change_event'] == category).astype(float)

    # Feature lists used in lags and rolls
    numerical_cols = ['gdp_growth_annualized', 'inflation_rate', 'central_bank_rate', 'political_stability_index', 'public_sentiment_score', 'macro_political_trend']

    # Re-calculate exact lag, roll, momentum features
    for col in numerical_cols:
        for lag in [1, 3, 7]:
            processed_df[f'{col}_lag_{lag}'] = processed_df[col].shift(lag)
        for window in [7, 30]:
            processed_df[f'{col}_roll_mean_{window}'] = processed_df[col].rolling(window=window).mean()
            processed_df[f'{col}_roll_std_{window}'] = processed_df[col].rolling(window=window).std()
            processed_df[f'{col}_roll_min_{window}'] = processed_df[col].rolling(window=window).min()
            processed_df[f'{col}_roll_max_{window}'] = processed_df[col].rolling(window=window).max()
            processed_df[f'{col}_momentum_{window}'] = processed_df[col] - processed_df[col].shift(window)

    # Keep only the rows that have non-null entries for features (typically the last row in inference stage)
    features_only = processed_df[feature_cols].iloc[-1:]

    # 2. Extract calibrated ensemble models for classification (7-day)
    with open('models/ensemble_calibrated_7d.pkl', 'rb') as f:
        ensemble_cls_7d = pickle.load(f)

    xgb_model_7d = ensemble_cls_7d['xgb_model']
    lgb_model_7d = ensemble_cls_7d['lgb_model']
    calibrator_7d = ensemble_cls_7d['calibrator']

    # Inference Predictions (7-Day Multi-Class probability)
    raw_xgb_prob = xgb_model_7d.predict_proba(features_only)
    raw_lgb_prob = lgb_model_7d.predict_proba(features_only)
    raw_ensemble_prob = (raw_xgb_prob + raw_lgb_prob) / 2.0

    calibrated_prob = calibrator_7d.predict_proba(raw_ensemble_prob)[0]
    predicted_class = np.argmax(calibrated_prob) - 1 # Map [0, 1, 2] back to [-1, 0, 1]

    # Classification Label mapping
    class_labels = { -1: 'Negative Shift (-1)', 0: 'Neutral (0)', 1: 'Positive Shift (+1)' }
    predicted_label = class_labels[predicted_class]

    # 3. Predict 30-Day continuous trend regression value

            'Positive_Prob': float(calibrated_prob[2])

import numpy as np

def audit_data_quality_engine(df):
    print("=== Starting Advanced Data Quality & Temporal Leakage Audit ===")
    df_audit = df.copy()
    df_audit['date'] = pd.to_datetime(df_audit['date'])
    df_audit = df_audit.sort_values('date').reset_index(drop=True)

    report = {}

    # 1. Validasi Kelengkapan Tanggal (Gap Detection)
    date_range = pd.date_range(start=df_audit['date'].min(), end=df_audit['date'].max(), freq='D')
    missing_dates = len(date_range) - len(df_audit['date'].unique())
    report['missing_dates_count'] = int(missing_dates)
    if missing_dates > 0:
        print(f"[WARNING] Temporal gaps detected! Missing {missing_dates} days in date range.")
    else:
        print("[OK] Temporal consistency passed. No daily gaps detected.")

    # 2. Deteksi Outlier Ekstrem (Metode IQR)
    outlier_features = ['inflation_rate', 'political_stability_index']
    report['outliers'] = {}
    for col in outlier_features:
        Q1 = df_audit[col].quantile(0.25)
        Q3 = df_audit[col].quantile(0.75)
        IQR = Q3 - Q1
        lower_bound = Q1 - 1.5 * IQR
        upper_bound = Q3 + 1.5 * IQR
        outliers_count = ((df_audit[col] < lower_bound) | (df_audit[col] > upper_bound)).sum()
        report['outliers'][col] = int(outliers_count)
        print(f"[INFO] Outliers detected in '{col}': {outliers_count} rows outside bounds [{lower_bound:.4f}, {upper_bound:.4f}].")

    # 3. Analisis Cross-Correlation untuk Audit Kebocoran Temporal
    # Mengecek apakah lag 0 fitur prediktif berkorelasi sangat tinggi dengan pergeseran masa depan target
    target_col = 'target_reg_30d'
    potential_predictors = ['macro_political_trend', 'inflation_rate', 'political_stability_index']
    report['cross_correlations_with_future_target'] = {}
    leakage_alert = False

    for pred in potential_predictors:
        # Menghitung korelasi pearson langsung
        corr_val = df_audit[pred].corr(df_audit[target_col])
        report['cross_correlations_with_future_target'][pred] = float(corr_val)
        print(f"[INFO] Correlation between present '{pred}' and future target '{target_col}': {corr_val:.4f}")
        if abs(corr_val) > 0.90:
            print(f"[LEAKAGE WARNING] Extremely high correlation ({corr_val:.4f}) detected! Potential look-ahead bias.")
            leakage_alert = True

    report['leakage_alert_status'] = leakage_alert

    # Simpan laporan ke JSON
    os.makedirs('artifacts', exist_ok=True)
    with open('artifacts/comprehensive_data_audit.json', 'w') as f:
        json.dump(report, f, indent=4)

    print("=== Audit Completed. Report saved to 'artifacts/comprehensive_data_audit.json' ===\n")
    return report

import numpy as np
import pandas as pd
import os

def calculate_psi(expected, actual, num_buckets=10):
    """Calculates the Population Stability Index (PSI) between training (expected) and test (actual) distributions."""
    # Determine bin boundaries using percentiles from expected (train) dataset
    percentiles = np.linspace(0, 100, num_buckets + 1)
    buckets = np.percentile(expected, percentiles)
    buckets[0] = -np.inf
    buckets[-1] = np.inf

    # Prevent identical bucket boundaries due to low variance
    buckets = np.unique(buckets)
    if len(buckets) < 2:
        return 0.0

    # Calculate counts in each bucket
    expected_counts, _ = np.histogram(expected, bins=buckets)
    actual_counts, _ = np.histogram(actual, bins=buckets)

    # Convert to fractions with small offset to prevent division by zero or log(0) problems
    expected_pct = np.where(expected_counts == 0, 0.0001, expected_counts) / len(expected)
    actual_pct = np.where(actual_counts == 0, 0.0001, actual_counts) / len(actual)

    # Compute PSI
    psi_value = np.sum((actual_pct - expected_pct) * np.log(actual_pct / expected_pct))
    return psi_value

def check_and_trigger_retraining(y_actual, y_predicted, X_new_train, y_new_train, mae_threshold=0.048):
    """
    Memantau performa model dan melakukan retraining otomatis jika MAE melampaui batas toleransi.
    """
    # 1. Hitung MAE saat ini
    current_mae = mean_absolute_error(y_actual, y_predicted)

    print(f"MAE Saat Ini     : {current_mae:.5f}")
    print(f"Ambang Batas MAE : {mae_threshold:.5f}")

import numpy as np
import pandas as pd

import seaborn as sns

def calculate_psi(expected, actual, num_buckets=10):
    """Calculates the Population Stability Index (PSI) between training (expected) and test (actual) distributions."""
    percentiles = np.linspace(0, 100, num_buckets + 1)
    buckets = np.percentile(expected, percentiles)
    buckets[0] = -np.inf
    buckets[-1] = np.inf

    buckets = np.unique(buckets)
    if len(buckets) < 2:
        return 0.0

    expected_counts, _ = np.histogram(expected, bins=buckets)
    actual_counts, _ = np.histogram(actual, bins=buckets)

    expected_pct = np.where(expected_counts == 0, 0.0001, expected_counts) / len(expected)
    actual_pct = np.where(actual_counts == 0, 0.0001, actual_counts) / len(actual)

    psi_value = np.sum((actual_pct - expected_pct) * np.log(actual_pct / expected_pct))
    return psi_value

def calculate_psi(expected, actual, num_buckets=10):
    percentiles = np.linspace(0, 100, num_buckets + 1)
    buckets = np.percentile(expected, percentiles)
    buckets[0] = -np.inf
    buckets[-1] = np.inf
    buckets = np.unique(buckets)
    if len(buckets) < 2:
        return 0.0
    expected_counts, _ = np.histogram(expected, bins=buckets)
    actual_counts, _ = np.histogram(actual, bins=buckets)
    expected_pct = np.where(expected_counts == 0, 0.0001, expected_counts) / len(expected)
    actual_pct = np.where(actual_counts == 0, 0.0001, actual_counts) / len(actual)
    return np.sum((actual_pct - expected_pct) * np.log(actual_pct / expected_pct))
